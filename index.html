Profil Singkat PT JALUR DATA INDONESIA

PT JALUR DATA INDONESIA (JDI) didirikan dengan tekad kuat mereduksi kesenjangan digital di area pedesaan dan sub-urban. Petunjuk Penggunaan & Kode Backend Google Apps Script (Code.gs)

Panduan lengkap pengoperasian & integrasi Google Sheets

A. Cara Mengakses Dasbor Admin:

1. Pada halaman publik di pojok kiri paling bawah terdapat tombol rahasia transparan 10% (ikon kunci kecil).
2. Klik tombol tersebut untuk membuka jendela login.
3. Masukkan kata sandi default: adminjdi123.

B. Fitur Otomatis Jatuh Tempo & Cetak Struk:

1. Setiap penambahan pelanggan baru, Tanggal Jatuh Tempo dihitung otomatis 30 hari dari tanggal pasang.
2. Tombol Cetak Nota langsung disesuaikan dengan format kertas thermal printer kasir 58mm / 80mm.
3. Terdapat tombol "Kirim Struk Terima Kasih via WA" yang membuka WhatsApp secara otomatis dengan pesan rapi dan rincian rekening transfer (Seabank, DANA, GoPay).

C. Pembuatan Database CSV (Wajib):

Masuk ke menu Database CSV & Sync, lalu klik "Buat Database Sekarang". Aplikasi akan meracik dan mengunduh database CSV terstandarisasi serta me-reload data secara otomatis.

Backend Google Apps Script (Code.gs)

Salin kode di bawah ini ke script.google.com, hubungkan dengan Google Spreadsheet Anda, lalu klik Deploy > New Deployment > Web App (Access: Anyone). 

/**
 * GOOGLE APPS SCRIPT BACKEND: PT JALUR DATA INDONESIA
 * Manajemen Pelanggan WIFI Keluar Masuk & Kas Keuangan
 */

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Pelanggan");
  if (!sheet) {
    sheet = ss.insertSheet("Pelanggan");
    sheet.appendRow(["ID", "Nama", "No_WA", "Alamat", "Paket", "Harga", "Tgl_Pasang", "Tgl_Tempo", "Status_Pelanggan", "Status_Bayar", "Foto"]);
  }
  
  var data = sheet.getDataRange().getValues();
  var headers = data[0];
  var result = [];
  
  for (var i = 1; i < data.length; i++) {
    var row = data[i];
    var obj = {};
    for (var j = 0; j < headers.length; j++) {
      obj[headers[j]] = row[j];
    }
    result.push(obj);
  }
  
  return ContentService.createTextOutput(JSON.stringify({
    status: "success",
    data: result
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("Pelanggan");
    if (!sheet) {
      sheet = ss.insertSheet("Pelanggan");
      sheet.appendRow(["ID", "Nama", "No_WA", "Alamat", "Paket", "Harga", "Tgl_Pasang", "Tgl_Tempo", "Status_Pelanggan", "Status_Bayar", "Foto"]);
    }
    
    var payload = JSON.parse(e.postData.contents);
    var action = payload.action;
    
    if (action === "syncAll") {
      sheet.clearContents();
      sheet.appendRow(["ID", "Nama", "No_WA", "Alamat", "Paket", "Harga", "Tgl_Pasang", "Tgl_Tempo", "Status_Pelanggan", "Status_Bayar", "Foto"]);
      
      var list = payload.data || [];
      for (var i = 0; i < list.length; i++) {
        var p = list[i];
        sheet.appendRow([
          p.id || "",
          p.nama || "",
          p.wa || "",
          p.alamat || "",
          p.paket || "",
          p.harga || 0,
          p.tglPasang || "",
          p.tglTempo || "",
          p.statusPelanggan || "Aktif",
          p.statusBayar || "Belum Lunas",
          p.foto || ""
        ]);
      }
      return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Sinkronisasi Google Sheets Berhasil!" })).setMimeType(ContentService.MimeType.JSON);
    }
    
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Aksi tidak dikenal" })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}
